if(USE_CROSS_COMPILER)
  set(RKNN_INCLUDE_DIR "" CACHE PATH "Path to directory containing rknn_api.h")
  set(RKNN_LIBRARY "" CACHE FILEPATH "Path to librknn_api.so or librknnrt.so")

  if(RKNN_INCLUDE_DIR STREQUAL "")
    if(EXISTS "${CMAKE_SYSROOT}/usr/include/rknn_api.h")
      set(RKNN_INCLUDE_DIR "${CMAKE_SYSROOT}/usr/include")
    elseif(EXISTS "$ENV{HOME}/sysroot/usr/include/rknn_api.h")
      set(RKNN_INCLUDE_DIR "$ENV{HOME}/sysroot/usr/include")
    endif()
  endif()

  if(RKNN_LIBRARY STREQUAL "")
    if(EXISTS "${CMAKE_SYSROOT}/usr/lib/librknn_api.so")
      set(RKNN_LIBRARY "${CMAKE_SYSROOT}/usr/lib/librknn_api.so")
    elseif(EXISTS "${CMAKE_SYSROOT}/usr/lib/librknnrt.so")
      set(RKNN_LIBRARY "${CMAKE_SYSROOT}/usr/lib/librknnrt.so")
    elseif(EXISTS "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu/librknn_api.so")
      set(RKNN_LIBRARY "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu/librknn_api.so")
    elseif(EXISTS "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu/librknnrt.so")
      set(RKNN_LIBRARY "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu/librknnrt.so")
    elseif(EXISTS "$ENV{HOME}/sysroot/usr/lib/librknn_api.so")
      set(RKNN_LIBRARY "$ENV{HOME}/sysroot/usr/lib/librknn_api.so")
    elseif(EXISTS "$ENV{HOME}/sysroot/usr/lib/librknnrt.so")
      set(RKNN_LIBRARY "$ENV{HOME}/sysroot/usr/lib/librknnrt.so")
    elseif(EXISTS "$ENV{HOME}/sysroot/usr/lib/aarch64-linux-gnu/librknn_api.so")
      set(RKNN_LIBRARY "$ENV{HOME}/sysroot/usr/lib/aarch64-linux-gnu/librknn_api.so")
    elseif(EXISTS "$ENV{HOME}/sysroot/usr/lib/aarch64-linux-gnu/librknnrt.so")
      set(RKNN_LIBRARY "$ENV{HOME}/sysroot/usr/lib/aarch64-linux-gnu/librknnrt.so")
    endif()
  endif()

  if(RKNN_INCLUDE_DIR STREQUAL "")
    message(FATAL_ERROR "RKNN_INCLUDE_DIR is required (directory containing rknn_api.h). Tried: ${CMAKE_SYSROOT}/usr/include and $ENV{HOME}/sysroot/usr/include")
  endif()
  if(RKNN_LIBRARY STREQUAL "")
    message(FATAL_ERROR "RKNN_LIBRARY is required (path to librknn_api.so/librknnrt.so). Tried: ${CMAKE_SYSROOT}/usr/lib and ${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu")
  endif()

  message(STATUS "RKNN_INCLUDE_DIR=${RKNN_INCLUDE_DIR}")
  message(STATUS "RKNN_LIBRARY=${RKNN_LIBRARY}")
  message("now using crosscompiler")
else()
  set(RKNN_INCLUDE_DIR "" CACHE PATH "Path to directory containing rknn_api.h")
  set(RKNN_LIBRARY "" CACHE FILEPATH "Path to librknn_api.so or librknnrt.so")

  if(RKNN_INCLUDE_DIR STREQUAL "")
    if(EXISTS "/usr/include/rknn_api.h")
      set(RKNN_INCLUDE_DIR "/usr/include" CACHE PATH "" FORCE)
    elseif(EXISTS "/usr/local/include/rknn_api.h")
      set(RKNN_INCLUDE_DIR "/usr/local/include" CACHE PATH "" FORCE)
    endif()
  endif()

  if(RKNN_INCLUDE_DIR STREQUAL "")
    find_path(RKNN_INCLUDE_DIR rknn_api.h
      PATHS /usr/include /usr/local/include
      PATH_SUFFIXES rknn rknn_api)
  endif()

  if(RKNN_LIBRARY STREQUAL "")
    if(EXISTS "/usr/lib/librknnrt.so")
      set(RKNN_LIBRARY "/usr/lib/librknnrt.so" CACHE FILEPATH "" FORCE)
    elseif(EXISTS "/usr/lib/aarch64-linux-gnu/librknnrt.so")
      set(RKNN_LIBRARY "/usr/lib/aarch64-linux-gnu/librknnrt.so" CACHE FILEPATH "" FORCE)
    elseif(EXISTS "/usr/lib/librknn_api.so")
      set(RKNN_LIBRARY "/usr/lib/librknn_api.so" CACHE FILEPATH "" FORCE)
    elseif(EXISTS "/usr/lib/aarch64-linux-gnu/librknn_api.so")
      set(RKNN_LIBRARY "/usr/lib/aarch64-linux-gnu/librknn_api.so" CACHE FILEPATH "" FORCE)
    endif()
  endif()

  if(RKNN_LIBRARY STREQUAL "")
    find_library(RKNN_LIBRARY
      NAMES rknnrt rknn_api
      PATHS /usr/lib /usr/local/lib /usr/lib/aarch64-linux-gnu /usr/local/lib/aarch64-linux-gnu)
  endif()

  if(RKNN_INCLUDE_DIR STREQUAL "")
    message(STATUS "RKNN_INCLUDE_DIR not found in standard system paths; RKNN targets will be disabled")
  endif()
  if(RKNN_LIBRARY STREQUAL "")
    message(STATUS "RKNN_LIBRARY not found in standard system paths; RKNN targets will be disabled")
  endif()

  if(NOT RKNN_INCLUDE_DIR STREQUAL "" AND NOT RKNN_LIBRARY STREQUAL "")
    message(STATUS "RKNN_INCLUDE_DIR=${RKNN_INCLUDE_DIR}")
    message(STATUS "RKNN_LIBRARY=${RKNN_LIBRARY}")
  endif()
  message("not using crosscompiler")
endif()
